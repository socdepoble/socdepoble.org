#!/usr/bin/env node
/**
 * Script de neteja de la brossa de l'arrel de Som de Poble i socdepoble.org
 * Envia els fitxers no llistats a la whitelist cap a .sdp-paperera per mantenir l'entorn de treball net.
 */
import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

const ROOT_SDP = path.resolve(__dirname, '../../');
const ROOT_WORKSPACE = path.resolve(ROOT_SDP, '../');

// Whitelist del repositori socdepoble.org
const ALLOWED_ROOT_SDP = [
  '.DS_Store',
  '.agents',
  '.env', '.env.example', '.env.local',
  '.git', '.github', '.gitignore', '.husky',
  '.sdp-paperera',
  '.vscode',
  'AGENTS.md', 'CLAUDE.md', 'INTEGRACIO.md', 'README.md', 'VEINS.md', 'LICENSE',
  'SEGELL.lock',
  '_templates', '_wiki_de_poble',
  'assets', 'bot', 'dist', 'docs', 'node_modules', 'public', 'requirements',
  'scratch', 'scripts', 'src', 'supabase', 'tests', 'tooling', 'tools', 'var',
  'wordpress-plugin',
  'docker-compose.yml', 'Dockerfile', '.dockerignore',
  'eslint.config.js',
  'index.html',
  'package-lock.json', 'package.json', 'pnpm-workspace.yaml',
  'vercel.json',
  'vite.config.js', 'vite.standalone.config.js'
];

// Whitelist del workspace Som de Poble
const ALLOWED_ROOT_WORKSPACE = [
  '.DS_Store',
  '.gitignore',
  '_arxiu_wiki_de_poble',
  '_comunicacio_de_poble',
  '_cultura_de_poble',
  '_gestoria_de_poble',
  '_multimedia_de_poble',
  '_raw_de_poble',
  'oficina_de_poble',
  'socdepoble.org'
];

function quarantine(root, file, paperera) {
  const src = path.join(root, file);
  const dest = path.join(paperera, file);
  try {
    if (fs.existsSync(dest)) {
      fs.rmSync(dest, { recursive: true, force: true });
    }
    fs.renameSync(src, dest);
    console.log(`[NETEJA] Enviat a quarantena: ${file} (des de ${path.basename(root)})`);
  } catch (e) {
    console.error(`[ERROR] No s'ha pogut moure ${file}: ${e.message}`);
  }
}

function cleanDir(dir, whitelist, paperera) {
  if (!fs.existsSync(dir)) return;
  const items = fs.readdirSync(dir);
  for (const item of items) {
    if (!whitelist.includes(item)) {
      quarantine(dir, item, paperera);
    }
  }
}

function main() {
  const paperera = path.join(ROOT_SDP, '.sdp-paperera', `quarantena-${new Date().toISOString().replace(/[:.]/g, '-')}`);
  fs.mkdirSync(paperera, { recursive: true });

  console.log(`🧹 Iniciant neteja de brossa...`);
  
  cleanDir(ROOT_WORKSPACE, ALLOWED_ROOT_WORKSPACE, paperera);
  cleanDir(ROOT_SDP, ALLOWED_ROOT_SDP, paperera);

  // Check if paperera is empty and remove if so
  if (fs.readdirSync(paperera).length === 0) {
    fs.rmdirSync(paperera);
    console.log(`✨ Tot estava net!`);
  } else {
    console.log(`🗑️ S'ha mogut la brossa a ${path.relative(ROOT_SDP, paperera)}`);
  }
}

main();
