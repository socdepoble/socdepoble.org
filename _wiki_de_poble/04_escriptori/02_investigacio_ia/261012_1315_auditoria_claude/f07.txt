/* El descodificador d'un recurs recorre el mateix camí que F3 de
   tractor-adaptadors.mjs valida amb la fixture: un recurs de llista passa per
   `extrau` (si en té) i per `traductor.llista`; la resta, per `traductor`. */
const descodificadorDe = (recurs) => {
  if (!recurs || typeof recurs !== 'object') return null;
  if (!recurs.llista) return typeof recurs.traductor === 'function' ? recurs.traductor : null;
  if (typeof recurs.traductor?.llista !== 'function') return null;
  return (payload) => recurs.traductor.llista(recurs.extrau ? recurs.extrau(payload) : payload);
};

/* La cola amb recursos.js, al patró de viquipedia/client.js: un descriptor de
   creaFrontissa per cada transport, amb el traductor registrat com a
   descodificador. Un transport sense traductor no entra (falla tancat) i el
   descodificador va l'últim: cap transport en pot posar un de propi. */
export const composaRecursos = (traductors, transports) => {
  const resources = {};
  for (const [nom, transport] of Object.entries(transports)) {
    const traductor = Object.hasOwn(traductors, nom) ? descodificadorDe(traductors[nom]) : null;
    if (typeof traductor !== 'function') throw new ErrorDades('INVALID_RESOURCE');
    resources[nom] = Object.freeze({ ...transport, decode: traductor });
  }
  return Object.freeze(resources);
};
