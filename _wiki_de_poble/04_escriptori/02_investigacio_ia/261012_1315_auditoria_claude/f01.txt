RESUM

entre grups, i el NIE (:65) i el NIF d'entitat (:66) exigixen set xifres seguides; IBAN, compte, afiliació i targeta no coneixen el punt.
Un número escrit «dd ddd ddd L», «dd-ddd-ddd-L» o «L-dd.ddd.ddd» (d = xifra, L = lletra) dona identificadorsDe = [] i passa per àlies,
per text lliure i per id opac. Importa per a la Finestreta perquè l'expedient (projectes/nucli/expedient.js:176-179) valida l'id, l'àlies
i el codi comptable amb estes mateixes guardes. Hui no hi ha fuga: el carril no està endollat i el Bar passa titular: null.
Es proposa admetre el separador opcional entre TOTS els grups de xifres de set formes, a guardes.js i a la còpia de registreVendes.js,
amb una prova nova i quatre files de paritat. El canvi té un cost de fals positiu que ha de decidir el Mestre; si diu que no, queda
escrita, completa, la variant agrupada.

import { describe, expect, test } from 'vitest';
import { validaDocument } from './document.js';
import { comprovaText, esIdOpac, identificadorsDe } from './guardes.js';
import { validaTitular } from './titular.js';

const tros = (...parts) => parts.join('');
const codis = (errors) => errors.map((e) => e.codi);

const LLAR = { id: 'tit_prova_llar', tipus: 'persona', ambit: 'LLAR', alies: 'Casa de prova' };
const RETA = {
  id: 'doc_prova_reta',
  titular: 'tit_prova_activitat',
  tipus: 'reta-alta-baixa',
  organisme: 'TGSS',
  dates: { document: '2026-01-10', notificacio: '2026-01-12', efectes: '2026-01-01' },
  referencia: 'ref_prova_001',
  sha256: 'AB'.repeat(32),
  teCodiVerificacio: true
};

describe('els separadors habituals compten entre tots els grups de xifres', () => {
  test.each([
    ['DNI', tros('12 ', '345 ', '678 ', 'Z')],
    ['DNI', tros('12-', '345-', '678-', 'Z')],
    ['DNI', tros('12.', '345.', '678', 'Z')],
    ['NIE', tros('X', '1 ', '234 ', '567 ', 'Z')],
    ['NIE', tros('X', '1.', '234.', '567', 'Z')],
    ['NIF_ENTITAT', tros('G', '12.', '345.', '678')],
    ['NIF_ENTITAT', tros('B-', '12.', '345.', '678')],
    ['NIF_ENTITAT', tros('G', '12 ', '345 ', '678')],
    ['IBAN', tros('ES91.', '2100.', '0418.', '4502.', '0005.', '1332')],
    ['COMPTE', tros('2100.', '0418.', '45.', '0200051332')],
    ['AFILIACIO', tros('28.', '1234', '5678.', '90')],
    ['TARGETA', tros('4000.', '0000.', '0000.', '0002')]
  ])('troba la forma %s', (forma, text) => {
    expect(identificadorsDe(text)).toContain(forma);
    expect(identificadorsDe(`Text de prova amb ${text} al mig`)).toContain(forma);
  });

  test('un id amb el número partit per guions o guions baixos tampoc és opac', () => {
    expect(esIdOpac(tros('fac-', '12-', '345-', '678-', 'z'))).toBe(false);
    expect(esIdOpac(tros('ref_', '12_', '345_', '678_', 'z'))).toBe(false);
  });

  test('el titular, el document i un text lliure refusen el número amb separadors', () => {
    expect(codis(validaTitular({ ...LLAR, alies: tros('Casa ', '12 ', '345 ', '678 ', 'Z') }))).toEqual(['IDENTIFICADOR_PERSONAL']);
    expect(codis(validaDocument({ ...RETA, referencia: tros('ref-', '12-', '345-', '678-', 'z') }))).toEqual(['ID_INVALID']);
    const errors = [];
    comprovaText(tros('CIF ', 'B-', '12.', '345.', '678'), 'alies', errors);
    expect(codis(errors)).toEqual(['IDENTIFICADOR_PERSONAL']);
  });
});
