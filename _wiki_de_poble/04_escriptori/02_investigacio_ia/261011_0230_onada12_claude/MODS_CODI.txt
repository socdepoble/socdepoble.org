D1 · CONTRACTE `sdp.mod.v1`
====================================================================

Què és. Un fitxer `tooling/mods/mod-<nom>.mjs`, Node pur i sense dependències, que exporta `MANIFEST` (congelat) i
una funció pura que torna l'informe. És l'equivalent, del costat del disc, d'un sentinella `sdp.sentinella.v1`
(`src/sections/tractors/sentinelles/contracte.js`), i en copia el vocabulari perquè la casa no n'aprenga un altre.

Manifest (tots obligatoris): `schema: 'sdp.mod.v1'`; `id` (`mod.<nom>`); `nom`; `versio` (semver); `nivell` (`L0`
només llig; `L1` pot demanar a una porta existent que moga coses que es poden desfer; cap `L2`: res que esborre);
`descripcio`; `fitxa` (ruta del Markdown d'aprovació); `fonts` (els fitxers que formen l'empremta: ell mateix i el
que importa); `llig` (abast de lectura, per superfície, mai per patró obert); `escriu` (per defecte `[]`; un L1 diu
quina porta escriu per ell i on); `xarxa: 'cap'`; `dadesPersonals`; `disparador`; `ordres` (les línies d'ordre
exactes, argv literal); `proves`.

Regles de comportament:
1. Sense bandera, o amb `--sec`, simula: imprimix el pla i no toca res. És el mode per defecte.
2. Actuar és `--aplica --codi CODI`, on el codi són els 12 primers caràcters del sha256 del pla que ha imprés
   `--sec`. Si la taula ha canviat, el codi no casa i no fa res (el mateix mecanisme que
   `cron_higiene.mjs --l2 --confirma`, `:425-434, 460`).
3. Un Mod no mou, no reanomena i no esborra amb `fs`: per a moure crida la porta que ja existix. Un Mod no escriu
   l'índex.
4. Eixida: un sol JSON per stdout (`schema: 'sdp.mod.informe.v1'`). Codis: 0 net, 1 hi ha troballes o feina, 2 no
   ha pogut auditar. Un abast absent és un 2, mai un verd.
5. Arrel per `arrelSegura()` i guarda `esMain(import.meta.url)` de `tooling/lib/arrel.mjs:284,360-364` (resol
   espais i enllaços simbòlics). Mai `process.cwd()`.
6. No llig el contingut de `90_arxiu_historic/` (AGENTS §5): com la porta, només noms. No entra a `01_produccio/`,
   `02_investigacio_ia/` ni a la Safata.

const RE_FILA = /^\s*\('([^']*)',\s*'([A-Za-z]+)',\s*'(.*)'::jsonb,\s*(\d+)\)[,;]?\s*$/;

/** Les files d'app_content de seed.sql: { clau, linia, versio, payload }. Pura. */
export function filesDeLaLlavor(sql) {
  const files = [];
  const illegibles = [];
  String(sql).split('\n').forEach((linia, i) => {
    const m = RE_FILA.exec(linia);
    if (!m) return;
    try { files.push({ clau: m[2], linia: i + 1, versio: Number(m[4]), payload: JSON.parse(m[3].replaceAll("''", "'")) }); }
    catch (error) { illegibles.push({ clau: m[2], linia: i + 1, motiu: error.message }); }
  });
  return { files, illegibles };
}

/** Les rutes /assets/… amb extensió que nomena un text, sense consulta ni fragment. Pura. */
export function recursosDe(text) {
  const trobats = new Set();
  for (const [cru] of String(text).matchAll(RE_RECURS)) {
    if (cru.includes('$') || cru.includes('{')) continue;   // tros d'una plantilla: no és una ruta sencera
    const net = cru.split(/[?#]/)[0].replace(/[.:]+$/, '');
    if (/\.[A-Za-z0-9]{2,5}$/.test(net)) trobats.add(net);
  }
  return [...trobats].sort();
}

/* La ruta és un fitxer de public/ amb la caixa exacta de cada tram: a macOS el disc no la
   distingix i a Linux (la CI, el servidor) sí. Només readdir i stat: cap contingut. */
function creaCens(arrel) {
  const dirs = new Map();
  const noms = (abs) => {
    if (!dirs.has(abs)) {
      let llista = null;
      try { llista = new Set(fs.readdirSync(abs)); } catch { /* no és un directori llegible */ }
      dirs.set(abs, llista);
    }
    return dirs.get(abs);
  };
  return (ruta) => {
    let relatiu;
    try { relatiu = decodeURIComponent(ruta); } catch { return false; }
    const trams = relatiu.split('/').filter(Boolean);
    if (trams.some((tram) => tram === '.' || tram === '..')) return false;
    let cursor = path.join(arrel, PUBLIC);
    for (const tram of trams) {
      if (!noms(cursor)?.has(tram)) return false;
      cursor = path.join(cursor, tram);
    }
    try { return fs.statSync(cursor).isFile(); } catch { return false; }
  };
}

export function audita(arrel) {

  const colleccions = new Map();

  const sql = llig(path.join(arrel, SQL));
  if (sql === null) {
    troballes.push(troballa('LLV-FONT-ABSENT', 'groga', `${SQL} no és al disc: la llavor SQL no s'ha auditat.`, SQL));

    const { files, illegibles } = filesDeLaLlavor(sql);

    }

      if (!Array.isArray(f.payload)) continue;
      const font = `${SQL}:${f.linia}`;
      colleccions.set(f.clau, f);
      comptes.colleccions += 1;
      comptes.entrades += f.payload.length;
      const perId = new Map();
      const perTitol = new Map();
      f.payload.forEach((entrada, i) => {
        const id = entrada?.id;
        if (id === undefined || id === null || id === '') {
          troballes.push(troballa('LLV-SENSE-ID', 'groga', `«${f.clau}»[${i}] no porta id: el client la descarta en pintar (byId).`, font));
          return;
        }
        perId.set(String(id), [...(perId.get(String(id)) ?? []), i]);
        const titol = typeof entrada.title === 'string' ? entrada.title.trim() : '';
        if (titol) perTitol.set(titol, [...(perTitol.get(titol) ?? []), String(id)]);
      });
      for (const [id, llocs] of perId) {
        if (llocs.length > 1) troballes.push(troballa('LLV-ID-REPETIT', 'roja', `«${f.clau}» porta l'id «${id}» ${llocs.length} vegades (posicions ${llocs.join(', ')}).`, font));
      }
      for (const [titol, ids] of perTitol) {
        if (ids.length > 1) troballes.push(troballa('LLV-TITOL-REPETIT', 'groga', `«${f.clau}» porta el títol «${titol}» en ${ids.length} entrades (${ids.join(', ')}).`, font));
      }
    }

    /* Derivades: el mateix element en dues col·leccions, o un recurs en dues entrades. */
    const media = colleccions.get('mediaItems');
    if (media) {
      const font = `${SQL}:${media.linia}`;
      const titols = new Map();
      const titolsDe = (clau) => {
        if (!titols.has(clau)) {
          titols.set(clau, new Map((colleccions.get(clau)?.payload ?? [])
            .filter((e) => typeof e?.title === 'string').map((e) => [e.title.trim(), String(e.id)])));
        }

        }

      }

  const resum = { roja: 0, groga: 0, info: 0 };
  for (const t of troballes) resum[t.severitat] += 1;
  return {
    schema: 'sdp.mod.informe.v1', mod: MANIFEST.id, versio: MANIFEST.versio, mode: 'lectura',
    abast: { sql: SQL, public: PUBLIC, fonts: FONTS_JS }, comptes, resum,

 * mod-neteja-escriptori.mjs — Mod de batuda de l'Escriptori «sdp.mod.v1», nivell L1.
 *
 * Sense bandera o amb --sec: classifica cada document del primer nivell de 04_escriptori/
 * (viu amb motiu, viu caducat, consumit, sense àncora, ancorat sense declarar), les àncores
 * fantasma, la Safata i el sediment de l'índex, i proposa el pla. No toca res.
 *
 * Amb --aplica --codi CODI: executa EXACTAMENT el pla que ha imprés --sec, i només d'una
 * manera: cridant `porta-olivera.mjs --arxiva --nom …` (primer amb --sec). No mou, no
 * reanomena, no esborra ni escriu l'índex amb fs. Mai toca 01_produccio/, 02_investigacio_ia/
 * ni la Safata: la porta rebutja qualsevol nom amb barra. Mai crida el Somiador.
 *
 * Què NO decidix: un declarat viu amb motiu vàlid no ix mai; un declarat viu caducat només
 * ix amb --inclou-caducats, si és d'una jornada anterior i la seua categoria ix per nom.
 * L'edat sola no classifica res (no hi ha purga per antiguitat).
 *
 * Eixida: un JSON per stdout. Codi 0 res a fer; 1 hi ha pla, decisions o sediment (o la
 * porta no ha pogut moure-ho tot); 2 error d'ús, d'estructura o codi de pla que no casa.
 */

const PORTA = 'tooling/gates/porta-olivera.mjs';
const ORDRE = `node ${PORTA}`;
const INDEX = CAMINS.indexEscriptori;
const NOM_INDEX = path.posix.basename(INDEX);
const BLOQUEJOS = Object.freeze(['.agents/runtime/document-create.lock', '.agents/runtime/mutation.lock', '.agents/runtime/mutation.pending.json']);

const sha = (text) => createHash('sha256').update(text).digest('hex');

/** Cada document solt del primer nivell amb la seua classe i el perquè. Només llig. */
export function classifica(arrel, { dia = avui(), ara = Date.now() } = {}) {
  const conv = checkConvocatoria(arrel, { dia, ara });
  const vida = new Map(conv.vida.map((v) => [v.nom, v]));
  const consumits = new Set(ranci(arrel, dia).map((d) => d.nom));
  const anc = ancoratge(arrel);
  const senseAncora = new Set(anc.senseAncora);
  const perNom = new Map(artefactes(arrel).map((d) => [d.nom, d]));
  const documents = [...nomsSolts(arrel)].filter((nom) => nom !== NOM_INDEX).sort().map((nom) => {
    const art = perNom.get(nom) ?? null;
    const v = vida.get(nom) ?? null;
    const base = {
      nom, categoria: art?.categoria ?? null, jornada: art?.data ?? null,
      ixPerNom: Boolean(art) && (TRANSITORIES.test(art.categoria) || PER_NOM.test(art.categoria))
    };
    if (v?.valid) return { ...base, classe: 'viu', motiu: v.motiu };
    if (v) return { ...base, classe: 'viu-caducat', motiu: `${v.codi}: ${v.motiu}` };
    if (consumits.has(nom)) return { ...base, classe: 'consumit', motiu: 'categoria de torn, de jornada anterior i sense declaració de viu' };
    if (senseAncora.has(nom)) return { ...base, classe: 'sense-ancora', motiu: `no té cap àncora a ${NOM_INDEX}` };
    return { ...base, classe: 'ancorat', motiu: art ? 'ancorat fora d’«Activitat en curs»: la porta no el jutja' : 'nom fora de la taxonomia aammdd_hhmm_categoria_titol.ext' };
  });
  const fantasmes = remeis(arrel, anc.fantasma).map((r) => ({ ancora: r.ref, trobats: r.trobats, remei: r.ancora }));
  return {
    documents, fantasmes, safata: safata(arrel),
    convocatoria: { fase: conv.fase, troballes: conv.troballes.map((t) => `[${t.codi}] ${t.missatge}`) }
  };
}

/** El sediment de l'índex, mesurat, i què es pot compactar. Només llig l'índex; de la resta, noms. */
export function sedimentIndex(arrel) {
  const text = fs.readFileSync(path.join(arrel, INDEX), 'utf8');
  const linies = text.split('\n');
  if (text.endsWith('\n')) linies.pop();
  const prosa = prosaIndex(text).split('\n');
  const caps = [];
  linies.forEach((_, i) => {
    const m = /^##[ \t]+(.+?)[ \t]*$/.exec(prosa[i] ?? '');
    if (m) caps.push({ titol: m[1], linia: i + 1 });
  });
  const seccions = caps.map((c, k) => {
    const final = (caps[k + 1]?.linia ?? linies.length + 1) - 1;
    const cos = prosa.slice(c.linia, final);
    return {
      titol: c.titol, linia: c.linia, final, linies: final - c.linia + 1,
      ancores: (cos.join('\n').match(/!?\[\[[^[\]\n]+\]\]/g) ?? []).length,
      blancs: cos.filter((l) => !l.trim()).length,
      buida: cos.every((l) => !l.trim())
    };
  });
  let blancsSobrers = 0;
  for (let i = 1; i < linies.length; i += 1) if (!linies[i].trim() && !linies[i - 1].trim()) blancsSobrers += 1;

  const perDesti = { taula: 0, laboratori: 0, produccio: 0, arxiu: 0, actes: 0, altres: 0 };
  const absents = [];
  const escriptori = path.join(arrel, CAMINS.escriptori);
  const hiEs = (abs) => { try { fs.lstatSync(abs); return true; } catch { return false; } };
  for (const a of ancoresDe(text)) {
    const ref = a.raw.replace(/\\(?=[|#])/g, '').split(/[|#]/)[0].trim();
    const classe = !ref.includes('/') ? 'taula'
      : ref.startsWith('02_investigacio_ia/') ? 'laboratori'
        : ref.startsWith('01_produccio/') ? 'produccio'
          : ref.startsWith('../90_arxiu_historic/') ? 'arxiu'
            : ref.startsWith('../10_actes/') ? 'actes' : 'altres';
    perDesti[classe] += 1;
    if (classe === 'taula') continue;   // les del primer nivell les jutja la porta (ancoratge)
    // Només existència (lstat): mai es llig el contingut, tampoc el de l'arxiu (AGENTS §5).
    const abs = path.resolve(escriptori, ref);
    if (!hiEs(abs) && !hiEs(`${abs}.md`)) absents.push(ref);
  }

  /* Declaracions d'«Activitat en curs» que apunten fora de la taula: la porta no les veu
     (només resol el primer nivell) i no caduquen mai. */
  const foraDeTaula = [];
  for (const s of seccions.filter((x) => /^Activitat en curs/.test(x.titol))) {
    for (let i = s.linia; i < s.final; i += 1) {
      const m = /^[ \t]*[-*][ \t]+!?\[\[([^[\]\n]+)\]\]/.exec(prosa[i] ?? '');
      if (m && m[1].split(/[|#]/)[0].includes('/')) foraDeTaula.push({ linia: i + 1, ancora: m[1] });
    }
  }

  const buides = seccions.filter((s) => s.buida);
  const fix = seccions.find((s) => /^Fix Automàtic/i.test(s.titol)) ?? null;
  const arxivats = seccions.find((s) => s.titol === SECCIO_ARXIVATS) ?? null;
  const mesura = sediment(arrel);
  return {
    linies: linies.length, llindar: LLINDAR_SEDIMENT, bytes: mesura.bytes, saturat: linies.length > LLINDAR_SEDIMENT,
    seccions: seccions.length, blancsSobrers,
    seccionsBuides: buides.map((s) => ({ titol: s.titol, linia: s.linia })),
    blocFixAutomatic: fix ? { linia: fix.linia, final: fix.final, ancores: fix.ancores, blancs: fix.blancs } : null,
    arxivats: arxivats ? { linia: arxivats.linia, final: arxivats.final, ancores: arxivats.ancores } : null,
    ancoresPerDesti: perDesti, ancoresAbsents: absents, declaracionsForaDeTaula: foraDeTaula,
    compactacio: {
      senseTocarCapAncora: {
        liniesEnBlancSobreres: blancsSobrers,
        seccionsBuides: buides.length,
        liniesEstimadesDespres: Math.max(0, linies.length - blancsSobrers - buides.length),
        qui: 'l’orquestradora, amb l’eina d’edició; cap enllaç es perd'
      },
      ambDecisioHumana: [
        { que: `${perDesti.laboratori + perDesti.produccio} àncores a 02_investigacio_ia/ i 01_produccio/ cap a un índex per carpeta`, perque: 'la porta SCC només acredita les àncores escrites a 00_index_escriptori.md (verificador-scc.mjs, verifyEscriptoriAnchoring): abans cal que seguisca un nivell de subíndexs' },
        { que: `${arxivats?.ancores ?? 0} línies de «${SECCIO_ARXIVATS}»`, perque: 'només es poden llevar quan el torn de cada una està tancat, i la línia no diu de quin torn és' },
        { que: `${foraDeTaula.length} declaracions d’«Activitat en curs» que apunten fora de la taula`, perque: 'porten motius del Mestre i de torns: decidir si continuen és seu' },
        { que: fix ? `bloc «Fix Automàtic d’Àncores» (línies ${fix.linia}-${fix.final}, ${fix.ancores} àncores, ${fix.blancs} línies en blanc)` : 'cap bloc «Fix Automàtic»', perque: 'les seues àncores són de 01_produccio/: fondre-les amb la secció «01_produccio» és un gest sobre la zona humana' }
      ]
    }
  };
}

/** El pla: què s'arxivaria, amb quina ordre de la porta, i què queda per decidir. Només llig. */
export function pla(arrel, { dia = avui(), ara = Date.now(), inclouCaducats = false } = {}) {
  const c = classifica(arrel, { dia, ara });
  const consumits = c.documents.filter((d) => d.classe === 'consumit').map((d) => d.nom);
  const caducats = c.documents
    .filter((d) => d.classe === 'viu-caducat' && d.ixPerNom && d.jornada !== null && d.jornada < dia).map((d) => d.nom);
  const arxiva = [...consumits, ...(inclouCaducats ? caducats : [])].sort();
  const decisions = [];
  for (const d of c.documents) {
    if (arxiva.includes(d.nom)) continue;
    if (d.classe === 'viu-caducat') {
      decisions.push({ sobre: d.nom, motiu: d.motiu, opcions: [
        'dona-li motiu a la seua línia d’«Activitat en curs» (· per a [[consumidor]], · torn_ID o · Mestre aammdd: motiu)',
        d.ixPerNom ? `${ORDRE} --arxiva --sec --nom ${d.nom}` : 'la seua categoria no ix per nom: pregunta al Mestre'
      ] });
    } else if (d.classe === 'sense-ancora') {
      decisions.push({ sobre: d.nom, motiu: d.motiu, opcions: [`ancora’l a ${NOM_INDEX}; si ja està consumit, ${ORDRE} --arxiva --sec --nom ${d.nom}`] });
    }
  }
  for (const f of c.fantasmes) {
    decisions.push({
      sobre: `[[${f.ancora}]]`,
      motiu: f.remei ? `àncora fantasma: el document és a ${f.trobats[0]}` : f.trobats.length ? `àncora fantasma ambigua: ${f.trobats.join(', ')}` : 'àncora fantasma sense destí als llocs que cerca la porta',
      opcions: [`${ORDRE} --repara --sec`]
    });
  }
  if (c.safata.elements.length) {
    decisions.push({ sobre: c.safata.carpeta, motiu: `${c.safata.elements.length} element(s) a la Safata d’Entrada`, opcions: ['classifica cada element: la porta només arxiva per nom documents del primer nivell, i este Mod no entra a la Safata'] });
  }
  const codi = sha(JSON.stringify({ dia, arxiva })).slice(0, 12);
  const noms = arxiva.flatMap((nom) => ['--nom', nom]).join(' ');
  const index = sedimentIndex(arrel);
  return {
    schema: 'sdp.mod.informe.v1', mod: MANIFEST.id, versio: MANIFEST.versio, mode: 'sec', dia,
    saturat: index.saturat || arxiva.length > 0 || c.documents.some((d) => d.classe === 'viu-caducat'),
    documents: c.documents, convocatoria: c.convocatoria,
    pla: {
      arxiva, codi,
      simulacio: arxiva.length ? `${ORDRE} --arxiva --sec ${noms}` : null,
      aplicacio: arxiva.length ? `node tooling/mods/mod-neteja-escriptori.mjs --aplica --codi ${codi}${inclouCaducats ? ' --inclou-caducats' : ''}` : null,
      caducatsQueEixirienAmbLaBandera: inclouCaducats ? [] : caducats
    },
    decisions, index
  };
}

/** Executa el pla vigent, i només eixe, a través de la porta Olivera. */
export function aplica(arrel, { codi, sessio = null, dia = avui(), ara = Date.now(), inclouCaducats = false } = {}) {
  const p = pla(arrel, { dia, ara, inclouCaducats });
  if (!p.pla.arxiva.length) return { ...p, mode: 'aplica', porta: null, motiu: 'Res a arxivar.' };
  if (codi !== p.pla.codi) throw new Error(`El codi no és el del pla vigent (${p.pla.codi}): la taula ha canviat o el pla no s'ha llegit. Torna a passar el mode --sec.`);
  const bloqueig = BLOQUEJOS.find((rel) => fs.existsSync(path.join(arrel, rel)));
  if (bloqueig) throw new Error(`${bloqueig} present: hi ha una escriptura documental o una mutació en curs. No s'ha mogut res.`);
  const args = [PORTA, '--arxiva', ...p.pla.arxiva.flatMap((nom) => ['--nom', nom]), ...(sessio ? ['--sessio', sessio] : [])];
  const opcions = { cwd: arrel, encoding: 'utf8', timeout: 60000 };
  const assaig = spawnSync(process.execPath, [...args, '--sec'], opcions);
  if (assaig.error || assaig.status !== 0) {
    throw new Error(`La simulació de la porta Olivera ha fallat; no s'ha mogut res: ${(assaig.error?.message ?? `${assaig.stderr}${assaig.stdout}`).trim()}`);
  }
  const real = spawnSync(process.execPath, args, opcions);
  return {
    ...p, mode: 'aplica',
    porta: { ordre: `node ${args.join(' ')}`, codi: real.status, eixida: `${real.stdout ?? ''}${real.stderr ?? ''}`.trim() },
    despres: classifica(arrel, { dia, ara: Date.now() }).documents.map(({ nom, classe }) => ({ nom, classe }))
  };
}

function opcionsDe(argv) {
  const o = { sec: false, aplica: false, codi: null, sessio: null, inclouCaducats: false };
  for (let i = 0; i < argv.length; i += 1) {
    const a = argv[i];
    if (a === '--sec') o.sec = true;
    else if (a === '--aplica') o.aplica = true;
    else if (a === '--inclou-caducats') o.inclouCaducats = true;
    else if (a === '--codi' || a === '--sessio') {
      const valor = argv[i + 1];
      if (!valor || valor.startsWith('--')) throw new Error(`${a} necessita un valor. ${US}`);
      o[a.slice(2)] = valor;
      i += 1;
    } else if (a === '--arrel') i += 1;
    else if (!a.startsWith('--arrel=')) throw new Error(`Argument desconegut: ${a}. ${US}`);
  }
  if (o.sec && o.aplica) throw new Error(`Tria --sec o --aplica. ${US}`);
  if (o.aplica && !/^[0-9a-f]{12}$/.test(o.codi ?? '')) throw new Error(`--aplica necessita el codi de 12 caràcters que dona el mode --sec. ${US}`);
  if (!o.aplica && (o.codi !== null || o.sessio !== null)) throw new Error(`--codi i --sessio només valen amb --aplica. ${US}`);
  if (o.sessio !== null && !SESSIO.test(o.sessio)) throw new Error(`--sessio ha de ser aammdd_Nom. ${US}`);
  return o;
}

function main(argv = process.argv.slice(2)) {
  const o = opcionsDe(argv);
  const arrel = arrelSegura();
  if (o.aplica) {
    const r = aplica(arrel, { codi: o.codi, sessio: o.sessio, inclouCaducats: o.inclouCaducats });
    process.stdout.write(`${JSON.stringify(r, null, 2)}\n`);
    return r.porta && r.porta.codi !== 0 ? 1 : 0;
  }
  const p = pla(arrel, { inclouCaducats: o.inclouCaducats });
  process.stdout.write(`${JSON.stringify(p, null, 2)}\n`);
  return p.saturat || p.decisions.length ? 1 : 0;
}

if (esMain(import.meta.url)) {
  try { process.exitCode = main(); }
  catch (error) {
    process.stdout.write(`${JSON.stringify({ schema: 'sdp.mod.informe.v1', mod: MANIFEST.id, valid: false, error: error.message })}\n`);
    process.exitCode = 2;
  }
}

  taula i es reancora a l'índex; si cal recuperar bytes d'una versió anterior, per `core-restauracio-segellada`.
  La porta no trepitja mai un destí, de manera que no hi ha res sobreescrit.
- Qualsevol canvi al Mod, a la porta Olivera o a esta fitxa invalida l'aprovació.
