### 3. Component Canònic: `PillToggle` (Selector de Píndola)

El Mestre vol preservar aquest patró com a element sagrat. L'extraiem com a component reutilitzable amb accessibilitat ARIA completa.

#### 3.1 Versió Web Component (per a `public/gestoria/tauler.js`)

```javascript
/**
 * PillToggle — Selector de vistes en píndola
 * Patró sagrat: [Universal Cards] [Vista Comprimida]
 * 
 * Ús:
 *   <pill-toggle options='["Universal Cards","Vista Comprimida"]'></pill-toggle>
 * 
 * Events:
 *   'pill-change' → detall: { index, value }
 */
class PillToggle extends HTMLElement {
  constructor() {
    super();
    this.attachShadow({ mode: 'open' });
    this._options = [];
    this._selected = 0;
  }

  static get observedAttributes() {
    return ['options', 'selected', 'aria-label'];
  }

  connectedCallback() {
    this._render();
    this.shadowRoot.querySelectorAll('.pill-toggle__btn').forEach(btn => {
      btn.addEventListener('click', (e) => this._onSelect(e));
    });
  }

  attributeChangedCallback(name, oldVal, newVal) {
    if (name === 'options') {
      try { this._options = JSON.parse(newVal); } catch { this._options = newVal.split(',').map(s => s.trim()); }
    }
    if (name === 'selected') this._selected = parseInt(newVal, 10) || 0;
    if (this.shadowRoot) this._render();
  }

  _onSelect(e) {
    const idx = parseInt(e.currentTarget.dataset.index, 10);
    if (idx === this._selected) return;
    this._selected = idx;
    this.setAttribute('selected', idx);
    this._render();
    this.dispatchEvent(new CustomEvent('pill-change', {
      detail: { index: idx, value: this._options[idx] },
      bubbles: true,
      composed: true
    }));
  }

  _render() {
    const label = this.getAttribute('aria-label') || 'Canviar vista';
    this.shadowRoot.innerHTML = `
      <style>
        .pill-toggle {
          display: inline-flex;
          align-items: center;
          gap: 2px;
          padding: 4px;
          background: var(--sdp-fons-targeta, #f8f6f4);
          border: 1px solid var(--sdp-pedra-300, #dcd7cd);
          border-radius: var(--sdp-radi-pill, 9999px);
        }
        .pill-toggle__btn {
          display: inline-flex;
          align-items: center;
          justify-content: center;
          padding: 8px 20px;
          min-height: 36px;
          border: none;
          background: transparent;
          color: var(--sdp-text-cos, #3d3b35);
          font-family: inherit;
          font-size: 0.875rem;
          font-weight: 500;
          line-height: 1;
          border-radius: var(--sdp-radi-pill, 9999px);
          cursor: pointer;
          transition: background-color 0.2s ease, color 0.2s ease, transform 0.1s ease;
          white-space: nowrap;
        }
        .pill-toggle__btn:hover {
          background: var(--sdp-pedra-150, #f3f0ec);
        }
        .pill-toggle__btn:active { transform: scale(0.97); }
        .pill-toggle__btn:focus-visible {
          outline: 2px solid var(--sdp-accio, #0984e3);
          outline-offset: 2px;
        }
        .pill-toggle__btn[aria-pressed="true"] {
          background: var(--sdp-accio, #0984e3);
          color: var(--sdp-blanc, #ffffff);
          box-shadow: var(--sdp-ombra-1, 0 1px 3px rgba(0,0,0,0.12));
        }
      </style>
      <div class="pill-toggle" role="group" aria-label="${label}">
        ${this._options.map((opt, i) => `
          <button 
            type="button"
            class="pill-toggle__btn"
            role="button"
            aria-pressed="${i === this._selected ? 'true' : 'false'}"
            data-index="${i}"
          >${opt}</button>
        `).join('')}
      </div>
    `;
    // Reattach listeners
    this.shadowRoot.querySelectorAll('.pill-toggle__btn').forEach(btn => {
      btn.addEventListener('click', (e) => this._onSelect(e));
    });
  }
}

customElements.define('pill-toggle', PillToggle);
```

#### 3.2 Versió React JSX (per a l'SPA principal)

```jsx
/**
 * PillToggle.jsx — Selector canònic de vistes en píndola
 * Preservació sagrada del patró del Mestre Javi
 */
import { useState } from 'react';

export function PillToggle({ 
  options, 
  defaultIndex = 0, 
  onChange, 
  ariaLabel = 'Canviar vista',
  className = ''
}) {
  const [selected, setSelected] = useState(defaultIndex);

  const handleSelect = (index) => {
    if (index === selected) return;
    setSelected(index);
    onChange?.(index, options[index]);
  };

  return (
    <div 
      className={`pill-toggle ${className}`}
      role="group" 
      aria-label={ariaLabel}
    >
      {options.map((option, index) => (
        <button
          key={`${option}-${index}`}
          type="button"
          className={`pill-toggle__btn ${index === selected ? 'is-active' : ''}`}
          role="button"
          aria-pressed={index === selected ? 'true' : 'false'}
          onClick={() => handleSelect(index)}
        >
          {option}
        </button>
      ))}
    </div>
  );
}
```
